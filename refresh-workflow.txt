name: Refresh Vuniper catalogues
on:
  workflow_dispatch:
  schedule:
    - cron: '17 */6 * * *'
permissions:
  contents: write
concurrency:
  group: vuniper-refresh
  cancel-in-progress: false
jobs:
  refresh:
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - uses: actions/setup-node@v4
        with:
          node-version: '22'
      - name: Fetch public lists
        run: python scripts/refresh.py
      - name: Validate catalogue and routes
        run: node --test
      - name: Commit fresh snapshot
        run: |
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          git add data/catalogues.json public/status.json
          git diff --cached --quiet || git commit -m 'Refresh Vuniper catalogues'
          git push
